យើងចង់ប្រើប្រាស់ event នៅក្នុង javascrip មកប្រើប្រាស់


<b [hidden]="hide">{{getSellers()}}</b>

<input type="button" value="Toggle" (click)="hide=!hide"/>


ចឹងបើគេចុតលើ button : Toggle ទៅបើវា បើកអោយបង្ហាញតម្លៃ 


--------------------------------------------

Example:

import { Component, OnInit } from '@angular/core';

@Component({
  selector: 'app-seller',
  templateUrl: './seller.component.html',
  styleUrls: ['./seller.component.css']
})
export class SellerComponent implements OnInit {

  sellers :String[];

  // create property 

  hide: boolean;

  constructor() { 
    this.sellers=["BestBuy","Apple","Amazon"],
    this.hide = true;
  }

  public getSellers():String[]{
    return this.sellers;
  }

  public toggle(){
    this.hide = !this.hide;
  }

  ngOnInit(): void {
  }

}


Example :
